func dnsName(nonce string) dnsmessage.Name {
    return dnsmessage.MustNewName(nonce+".apl-nativeqa.test.")
}
func answerDNS(input []byte, nonce string) ([]byte, bool) {
    if len(input)<12 || len(input)>512 {return nil,false}
    var q dnsmessage.Message
    if q.Unpack(input)!=nil || q.Response || q.OpCode!=0 || len(q.Questions)!=1 || len(q.Answers)!=0 || len(q.Authorities)!=0 {return nil,false}
    question:=q.Questions[0]
    if question.Class!=dnsmessage.ClassINET || !strings.EqualFold(question.Name.String(),dnsName(nonce).String()) {return nil,false}
    if question.Type!=dnsmessage.TypeA && question.Type!=dnsmessage.TypeAAAA {return nil,false}
    r:=dnsmessage.Message{Header:dnsmessage.Header{ID:q.ID,Response:true,Authoritative:true,RecursionDesired:q.RecursionDesired,RecursionAvailable:true},Questions:q.Questions}

    result,e:=r.Pack()

}
