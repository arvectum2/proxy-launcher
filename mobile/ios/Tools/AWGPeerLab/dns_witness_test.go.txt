package main
import ("testing";"golang.org/x/net/dns/dnsmessage";"strings")
func TestExactDNS(t *testing.T) {
 nonce:=strings.Repeat("a",48)
 q:=dnsmessage.Message{Header:dnsmessage.Header{ID:1},Questions:[]dnsmessage.Question{{Name:dnsName(nonce),Type:dnsmessage.TypeA,Class:dnsmessage.ClassINET}}}
 pack:=func()[]byte{b,e:=q.Pack();if e!=nil{t.Fatal(e)};return b}
 b,a:=answerDNS(pack(),nonce);var r dnsmessage.Message
 if !a||r.Unpack(b)!=nil||!r.Response||len(r.Answers)!=1||r.Answers[0].Body.(*dnsmessage.AResource).A!=[4]byte{192,0,2,2}{t.Fatal("A")}
 q.Questions[0].Type=dnsmessage.TypeAAAA;b,a=answerDNS(pack(),nonce)
 if a||r.Unpack(b)!=nil||len(r.Answers)!=0{t.Fatal("AAAA NODATA")}
 q.Questions[0].Name=dnsmessage.MustNewName("other.test.");b,_=answerDNS(pack(),nonce);if b!=nil{t.Fatal("other name")}
 q.Questions[0].Name=dnsName(nonce);q.Questions[0].Type=dnsmessage.TypeTXT;b,_=answerDNS(pack(),nonce);if b!=nil{t.Fatal("other type")}
}
